9  Dataset storage and versioning

42–50 minutes · pins

9.1 Introduction

We will store two versions of HealthWellbeing in a local pins board. The first contains all 29 studies. The second contains the 18 general-population studies, so we can explore how that restriction changes the analysis. Both use the variable definitions from the data dictionary, and both trace back to the same raw file. Dataset documentation

Work from the project root with box, pins, and the raw file from the preparation lesson. Runs reuse versions with the same data and provenance. The instructor can provide a prepared board and its selection files if you need to catch up. Prepare the raw file

9.2 Create a versioned board

We enable versioning so the board retains both writes. We use RDS, an R format that preserves the column types. The board lives inside our local workshop outputs.

box::use(
  pins[board_folder, pin_exists, pin_meta, pin_read, pin_versions,
       pin_write],
  utils[read.csv, write.csv]
)

workshop_dir <- file.path("outputs", "health-wellbeing")
data_board <- board_folder(
  file.path(workshop_dir, "data_store"),
  versioned = TRUE
)
pin_name <- "health_wellbeing"
versions_file <- file.path(workshop_dir, "dataset-versions.csv")
selection_file <- file.path(workshop_dir, "analysis-data-version.txt")

write_dataset_version <- function(board, name, data, description, metadata) {
  previous_versions <- character()
  if (pin_exists(board, name)) {
    previous_versions <- pin_versions(board, name)$version
    for (version in previous_versions) {
      if (identical(pin_meta(board, name, version = version)$user, metadata) &&
          identical(pin_read(board, name, version = version), data)) {
        return(version)
      }
    }
  }

  pin_write(
    board,
    data,
    name = name,
    type = "rds",
    title = "HealthWellbeing",
    description = description,
    metadata = metadata
  )
  new_version <- setdiff(pin_versions(board, name)$version, previous_versions)
  stopifnot(length(new_version) == 1L)
  new_version
}

9.3 Write and describe two versions

We verify the checksum, load the table from that file, and record its source and checksum in the pin metadata.

source_url <- paste0(
  "https://www.protectlab.org/meta-analysis-in-r/data/",
  "healthwellbeing.rda"
)
raw_file <- file.path("data", "raw", "healthwellbeing.rda")
raw_md5 <- unname(tools::md5sum(raw_file))
stopifnot(identical(raw_md5, "0d74964b0b7168379a2b015cb648b326"))

raw_data <- new.env(parent = baseenv())
loaded_names <- load(raw_file, envir = raw_data)
stopifnot(identical(loaded_names, "HealthWellbeing"))
verified_health <- raw_data[["HealthWellbeing"]]
stopifnot(is.data.frame(verified_health), nrow(verified_health) == 29L)

full_version <- write_dataset_version(
  data_board,
  pin_name,
  verified_health,
  description = "All 29 studies in the teaching snapshot.",
  metadata = list(source_url = source_url, raw_md5 = raw_md5, scope = "all")
)

For the teaching comparison, we read the exact full pin, then filter by population. The subset inherits its source and checksum from that pin. The filter and parent version account for the 11 excluded studies. The published example uses the full set.

full_metadata <- pin_meta(data_board, pin_name, version = full_version)
health_general <- pin_read(data_board, pin_name, version = full_version) |>
  subset(population == "general population")
stopifnot(nrow(health_general) == 18L)

general_version <- write_dataset_version(
  data_board,
  pin_name,
  health_general,
  description = "Only studies labelled general population.",
  metadata = list(
    source_url = full_metadata$user$source_url,
    raw_md5 = full_metadata$user$raw_md5,
    scope = "general",
    filter = "population == 'general population'",
    parent_version = full_version
  )
)

pin_versions(data_board, pin_name)
# A tibble: 2 × 3
  version                created             hash 
  <chr>                  <dttm>              <chr>
1 20261119T103524Z-11d00 2026-11-19 10:35:24 11d00

versions <- data.frame(
  scope = c("all", "general"),
  version = c(full_version, general_version)
)
write.csv(versions, versions_file, row.names = FALSE)
if (!file.exists(selection_file)) {
  writeLines(full_version, selection_file)
}

9.4 Try it yourself

Retrieve the full dataset by its recorded identifier and inspect its source metadata. The selection file starts with the full version and retains your choice on later runs.

versions <- read.csv(versions_file, colClasses = "character")
full_version <- versions$version[versions$scope == "all"]
stopifnot(length(full_version) == 1L)

health_full <- pin_read(data_board, pin_name, version = full_version)

9.5 Checkpoint

The history includes the full dataset and the subset under distinct identifiers. Retrieving the full version returns 29 rows. Locate the identifier of the 18-study subset in dataset-versions.csv. The version identifies a stored object within this board. If you recreate the board elsewhere, use its new identifiers and retain the upstream checksum and filter.

Keep the board local and share the source and transformation instructions when reuse terms are unclear. Next we will make the selection file a tracked pipeline input. Reproducible analytical pipelines
